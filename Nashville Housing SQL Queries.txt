/* -------------------------------------------------------
   1. GETTING STARTED - UNDERSTANDING THE DATA
------------------------------------------------------- */

-- First, I wanted to see what the dataset looks like.
SELECT *
FROM NashvilleHousing;


-- How many records are there?
SELECT COUNT(*) AS Total_Records
FROM NashvilleHousing;


-- How many unique properties are in the dataset?
SELECT COUNT(DISTINCT ParcelID) AS Unique_Properties
FROM NashvilleHousing;


/* -------------------------------------------------------
   2. CHECKING FOR MISSING VALUES
------------------------------------------------------- */

-- Checking important columns for NULL values.

SELECT
    SUM(CASE WHEN ParcelID IS NULL THEN 1 ELSE 0 END) AS Missing_ParcelID,
    SUM(CASE WHEN PropertyAddress IS NULL THEN 1 ELSE 0 END) AS Missing_PropertyAddress,
    SUM(CASE WHEN SalePrice IS NULL THEN 1 ELSE 0 END) AS Missing_SalePrice,
    SUM(CASE WHEN SaleDate IS NULL THEN 1 ELSE 0 END) AS Missing_SaleDate,
    SUM(CASE WHEN OwnerAddress IS NULL THEN 1 ELSE 0 END) AS Missing_OwnerAddress
FROM NashvilleHousing;


/* -------------------------------------------------------
   3. CLEANING THE SALE DATE
------------------------------------------------------- */

-- The original SaleDate column contains date and time.
-- I created a separate DATE column for easier analysis.

ALTER TABLE NashvilleHousing
ADD SaleDateConverted DATE;


UPDATE NashvilleHousing
SET SaleDateConverted - CONVERT(DATE, SaleDate);


-- Checking the result
SELECT
    SaleDate,
    SaleDateConverted
FROM NashvilleHousing;


/* -------------------------------------------------------
   4. FILLING MISSING PROPERTY ADDRESSES
------------------------------------------------------- */

-- Some properties have a missing PropertyAddress.
-- I used ParcelID to find another record for the same
-- property that contains the address.

SELECT
    a.ParcelID,
    a.PropertyAddress,
    b.PropertyAddress AS Matching_Address
FROM NashvilleHousing a
JOIN NashvilleHousing b
    ON a.ParcelID - b.ParcelID
    AND a.UniqueID <> b.UniqueID
WHERE a.PropertyAddress IS NULL
AND b.PropertyAddress IS NOT NULL;


-- Fill the missing addresses.

UPDATE a
SET a.PropertyAddress - b.PropertyAddress
FROM NashvilleHousing a
JOIN NashvilleHousing b
    ON a.ParcelID - b.ParcelID
    AND a.UniqueID <> b.UniqueID
WHERE a.PropertyAddress IS NULL
AND b.PropertyAddress IS NOT NULL;


/* -------------------------------------------------------
   5. SPLITTING PROPERTY ADDRESS
------------------------------------------------------- */

-- PropertyAddress contains both the street and city.
-- I separated them into two columns.

ALTER TABLE NashvilleHousing
ADD PropertySplitAddress NVARCHAR(255);

ALTER TABLE NashvilleHousing
ADD PropertySplitCity NVARCHAR(255);


UPDATE NashvilleHousing
SET PropertySplitAddress -
    SUBSTRING(
        PropertyAddress,
        1,
        CHARINDEX(',', PropertyAddress) - 1
    );


UPDATE NashvilleHousing
SET PropertySplitCity -
    SUBSTRING(
        PropertyAddress,
        CHARINDEX(',', PropertyAddress) + 1,
        LEN(PropertyAddress)
    );


-- Check the cleaned address
SELECT
    PropertyAddress,
    PropertySplitAddress,
    PropertySplitCity
FROM NashvilleHousing;


/* -------------------------------------------------------
   6. SPLITTING OWNER ADDRESS
------------------------------------------------------- */

-- OwnerAddress contains address, city and state.
-- I separated these into individual columns.

ALTER TABLE NashvilleHousing
ADD OwnerSplitAddress NVARCHAR(255);

ALTER TABLE NashvilleHousing
ADD OwnerSplitCity NVARCHAR(255);

ALTER TABLE NashvilleHousing
ADD OwnerSplitState NVARCHAR(50);


UPDATE NashvilleHousing
SET OwnerSplitAddress -
    PARSENAME(REPLACE(OwnerAddress, ',', '.'), 3);

UPDATE NashvilleHousing
SET OwnerSplitCity -
    PARSENAME(REPLACE(OwnerAddress, ',', '.'), 2);

UPDATE NashvilleHousing
SET OwnerSplitState -
    PARSENAME(REPLACE(OwnerAddress, ',', '.'), 1);


SELECT
    OwnerAddress,
    OwnerSplitAddress,
    OwnerSplitCity,
    OwnerSplitState
FROM NashvilleHousing;


/* -------------------------------------------------------
   7. CLEANING SOLD AS VACANT
------------------------------------------------------- */

-- First I checked the different values.

SELECT DISTINCT SoldAsVacant
FROM NashvilleHousing;


-- The column contains Y/N as well as Yes/No.
-- I standardized the values.

UPDATE NashvilleHousing
SET SoldAsVacant -
    CASE
        WHEN SoldAsVacant - 'Y' THEN 'Yes'
        WHEN SoldAsVacant - 'N' THEN 'No'
        ELSE SoldAsVacant
    END;


-- Check the result
SELECT DISTINCT SoldAsVacant
FROM NashvilleHousing;


/* -------------------------------------------------------
   8. FINDING DUPLICATE RECORDS
------------------------------------------------------- */

-- I used ROW_NUMBER() to identify possible duplicates.

WITH DuplicateRecords AS
(
    SELECT
        *,
        ROW_NUMBER() OVER
        (
            PARTITION BY
                ParcelID,
                PropertyAddress,
                SalePrice,
                SaleDate,
                LegalReference
            ORDER BY UniqueID
        ) AS RowNum
    FROM NashvilleHousing
)

SELECT *
FROM DuplicateRecords
WHERE RowNum > 1;

/* -------------------------------------------------------
   9. BASIC SALES ANALYSIS
------------------------------------------------------- */

-- Total number of sales
SELECT
    COUNT(*) AS Total_Sales
FROM NashvilleHousing;


-- Total sales value
SELECT
    SUM(SalePrice) AS Total_Sales_Value
FROM NashvilleHousing;


-- Average sale price
SELECT
    AVG(SalePrice) AS Average_Sale_Price
FROM NashvilleHousing;


-- Highest sale price
SELECT
    MAX(SalePrice) AS Highest_Sale_Price
FROM NashvilleHousing;


-- Lowest sale price
SELECT
    MIN(SalePrice) AS Lowest_Sale_Price
FROM NashvilleHousing
WHERE SalePrice > 0;


/* -------------------------------------------------------
   10. SALES BY YEAR
------------------------------------------------------- */

SELECT
    YEAR(SaleDateConverted) AS Sale_Year,
    COUNT(*) AS Number_of_Sales,
    SUM(SalePrice) AS Total_Sales,
    AVG(SalePrice) AS Average_Sale_Price
FROM NashvilleHousing
GROUP BY YEAR(SaleDateConverted)
ORDER BY Sale_Year;


/* -------------------------------------------------------
   11. SALES BY MONTH
------------------------------------------------------- */

SELECT
    YEAR(SaleDateConverted) AS Sale_Year,
    MONTH(SaleDateConverted) AS Sale_Month,
    COUNT(*) AS Number_of_Sales,
    SUM(SalePrice) AS Total_Sales
FROM NashvilleHousing
GROUP BY
    YEAR(SaleDateConverted),
    MONTH(SaleDateConverted)
ORDER BY
    Sale_Year,
    Sale_Month;


/* -------------------------------------------------------
   12. SALES BY PROPERTY TYPE
------------------------------------------------------- */

SELECT
    LandUse AS Property_Type,
    COUNT(*) AS Number_of_Properties,
    SUM(SalePrice) AS Total_Sales,
    AVG(SalePrice) AS Average_Sale_Price
FROM NashvilleHousing
GROUP BY LandUse
ORDER BY Total_Sales DESC;


/* -------------------------------------------------------
   13. SALES BY CITY
------------------------------------------------------- */

SELECT
    PropertySplitCity,
    COUNT(*) AS Number_of_Sales,
    SUM(SalePrice) AS Total_Sales,
    AVG(SalePrice) AS Average_Sale_Price
FROM NashvilleHousing
GROUP BY PropertySplitCity
ORDER BY Total_Sales DESC;


/* -------------------------------------------------------
   14. TOP 10 MOST EXPENSIVE PROPERTIES
------------------------------------------------------- */

SELECT TOP 10
    ParcelID,
    PropertySplitAddress,
    PropertySplitCity,
    SalePrice,
    SaleDateConverted
FROM NashvilleHousing
ORDER BY SalePrice DESC;


/* -------------------------------------------------------
   15. SALES ABOVE THE AVERAGE
------------------------------------------------------- */

SELECT
    ParcelID,
    PropertySplitAddress,
    PropertySplitCity,
    SalePrice
FROM NashvilleHousing
WHERE SalePrice >
(
    SELECT AVG(SalePrice)
    FROM NashvilleHousing
)
ORDER BY SalePrice DESC;


/* -------------------------------------------------------
   16. PROPERTY AGE AT THE TIME OF SALE
------------------------------------------------------- */

SELECT
    ParcelID,
    YEARBUILT,
    YEAR(SaleDateConverted) AS Sale_Year,
    YEAR(SaleDateConverted) - YEARBUILT AS Property_Age,
    SalePrice
FROM NashvilleHousing
WHERE YEARBUILT IS NOT NULL;


/* -------------------------------------------------------
   17. PRICE PER ACRE
------------------------------------------------------- */

SELECT
    ParcelID,
    Acreage,
    SalePrice,
    SalePrice / NULLIF(Acreage, 0) AS Price_Per_Acre
FROM NashvilleHousing
WHERE Acreage > 0;


/* -------------------------------------------------------
   18. SOLD AS VACANT ANALYSIS
------------------------------------------------------- */

SELECT
    SoldAsVacant,
    COUNT(*) AS Number_of_Properties,
    SUM(SalePrice) AS Total_Sales,
    AVG(SalePrice) AS Average_Sale_Price
FROM NashvilleHousing
GROUP BY SoldAsVacant
ORDER BY Number_of_Properties DESC;


/* -------------------------------------------------------
   19. RANKING PROPERTIES
------------------------------------------------------- */

-- Ranking properties based on their sale price.

SELECT
    ParcelID,
    PropertySplitAddress,
    PropertySplitCity,
    SalePrice,
    RANK() OVER (
        ORDER BY SalePrice DESC
    ) AS Price_Rank
FROM NashvilleHousing;


/* -------------------------------------------------------
   20. TOP 3 PROPERTIES WITHIN EACH CITY
------------------------------------------------------- */

WITH RankedProperties AS
(
    SELECT
        PropertySplitCity,
        ParcelID,
        PropertySplitAddress,
        SalePrice,
        RANK() OVER
        (
            PARTITION BY PropertySplitCity
            ORDER BY SalePrice DESC
        ) AS City_Rank
    FROM NashvilleHousing
)

SELECT *
FROM RankedProperties
WHERE City_Rank <- 3;


/* -------------------------------------------------------
   21. YEAR-OVER-YEAR SALES
------------------------------------------------------- */

WITH YearlySales AS
(
    SELECT
        YEAR(SaleDateConverted) AS Sale_Year,
        SUM(SalePrice) AS Total_Sales
    FROM NashvilleHousing
    GROUP BY YEAR(SaleDateConverted)
)

SELECT
    Sale_Year,
    Total_Sales,

    LAG(Total_Sales) OVER
    (
        ORDER BY Sale_Year
    ) AS Previous_Year_Sales,

    Total_Sales -
        LAG(Total_Sales) OVER
        (
            ORDER BY Sale_Year
        ) AS Sales_Change

FROM YearlySales
ORDER BY Sale_Year;


/* -------------------------------------------------------
   22. YEAR-OVER-YEAR GROWTH %
------------------------------------------------------- */

WITH YearlySales AS
(
    SELECT
        YEAR(SaleDateConverted) AS Sale_Year,
        SUM(SalePrice) AS Total_Sales
    FROM NashvilleHousing
    GROUP BY YEAR(SaleDateConverted)
),

SalesComparison AS
(
    SELECT
        Sale_Year,
        Total_Sales,

        LAG(Total_Sales) OVER
        (
            ORDER BY Sale_Year
        ) AS Previous_Year_Sales

    FROM YearlySales
)

SELECT
    Sale_Year,
    Total_Sales,
    Previous_Year_Sales,

    (
        (Total_Sales - Previous_Year_Sales)
        / NULLIF(Previous_Year_Sales, 0)
    ) * 100 AS Growth_Percent

FROM SalesComparison
ORDER BY Sale_Year;


/* -------------------------------------------------------
   23. RUNNING TOTAL OF SALES
------------------------------------------------------- */

WITH MonthlySales AS
(
    SELECT
        YEAR(SaleDateConverted) AS Sale_Year,
        MONTH(SaleDateConverted) AS Sale_Month,
        SUM(SalePrice) AS Monthly_Sales
    FROM NashvilleHousing
    GROUP BY
        YEAR(SaleDateConverted),
        MONTH(SaleDateConverted)
)

SELECT
    Sale_Year,
    Sale_Month,
    Monthly_Sales,

    SUM(Monthly_Sales) OVER
    (
        ORDER BY Sale_Year, Sale_Month
    ) AS Running_Total

FROM MonthlySales
ORDER BY
    Sale_Year,
    Sale_Month;


/* -------------------------------------------------------
   24. PROPERTY PRICE CATEGORIES
------------------------------------------------------- */

SELECT
    ParcelID,
    SalePrice,

    CASE
        WHEN SalePrice < 100000 THEN 'Below 100K'
        WHEN SalePrice < 250000 THEN '100K - 250K'
        WHEN SalePrice < 500000 THEN '250K - 500K'
        WHEN SalePrice < 1000000 THEN '500K - 1M'
        ELSE '1M+'
    END AS Price_Category

FROM NashvilleHousing;


/* -------------------------------------------------------
   25. NUMBER OF PROPERTIES BY PRICE CATEGORY
------------------------------------------------------- */

WITH PriceCategories AS
(
    SELECT
        CASE
            WHEN SalePrice < 100000 THEN 'Below 100K'
            WHEN SalePrice < 250000 THEN '100K - 250K'
            WHEN SalePrice < 500000 THEN '250K - 500K'
            WHEN SalePrice < 1000000 THEN '500K - 1M'
            ELSE '1M+'
        END AS Price_Category

    FROM NashvilleHousing
)

SELECT
    Price_Category,
    COUNT(*) AS Number_of_Properties

FROM PriceCategories
GROUP BY Price_Category
ORDER BY Number_of_Properties DESC;


/* -------------------------------------------------------
   26. PROPERTIES WITH MULTIPLE SALES
------------------------------------------------------- */

SELECT
    ParcelID,
    COUNT(*) AS Number_of_Sales
FROM NashvilleHousing
GROUP BY ParcelID
HAVING COUNT(*) > 1
ORDER BY Number_of_Sales DESC;


/* -------------------------------------------------------
   27. FINAL CLEAN DATASET
------------------------------------------------------- */

SELECT
    UniqueID,
    ParcelID,
    LandUse,
    PropertySplitAddress AS Property_Address,
    PropertySplitCity AS Property_City,
    SalePrice,
    SaleDateConverted AS Sale_Date,
    LegalReference,
    SoldAsVacant,
    OwnerSplitAddress AS Owner_Address,
    OwnerSplitCity AS Owner_City,
    OwnerSplitState AS Owner_State,
    Acreage,
    LandValue,
    BuildingValue,
    TotalValue,
    Bedrooms,
    FullBath,
    HalfBath,
    YEARBUILT
FROM NashvilleHousing;
